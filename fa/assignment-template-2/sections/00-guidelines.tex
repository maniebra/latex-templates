\begin{neobox}[neotitlefont=\pixelfont]{قوانین، نکات و سیاست‌ها}

    \vspace{5mm}
    \begin{center}
        {\large \textbf{پیش از حل سؤالات، موارد زیر را مطالعه فرمایید.}}
    \end{center}
    \vspace{5mm}

    \begin{enumerate}
        \item این آزمون \examgrade{} نمره دارد و نمره‌ی مازاد آن بر بقیه‌ی درس
              سرریز \underline{نخواهد کرد}.

        \item مدت آزمون \examtime{} دقیقه است و به هیچ‌وجه تمدید نخواهد شد. این
              زمان شامل مهلت آپلود پاسخ‌ها هم می‌شود.

        \item سیاست تأخیر را اینجا بنویسید؛ مثلاً به ازای هر نیم‌ساعت تأخیر،
              درصدی از نمره‌ی کل کسر می‌شود.

        \item استفاده از اسلایدهای درس، منابع معرفی‌شده و هوش مصنوعی برای
              \underline{مشورت} مجاز است. درج پاسخ دیگران یا پاسخ هوش مصنوعی
              در صورت کشف، نمره‌ی \underline{صفر} در پی دارد.

        \item پاسخ‌ها باید \underline{دست‌نویس} و در قالب \lr{PDF} ارسال شوند.

        \item {
              قالب فایل ارسالی شما به صورت زیر باشد:
              \begin{neocode}[neocodetitlefont=\small]{\lr{\pixellatin file\_format.txt}}
@\coursecode@-MID-[STDID].pdf
              \end{neocode}
              که در آن \lr{\texttt{STDID}} شماره‌ی دانشجویی شماست.
              }

        \item تنها پلتفرم مورد قبول برای ارسال پاسخ‌ها سامانه‌ی درس است؛ ایمیل و
              پیام‌رسان‌ها پذیرفته \underline{نمی‌شوند}.

        \item پاسخ‌ها باید خوانا نوشته و تصویربرداری شوند. دقیق‌نویسی و
              کامل‌نویسی خودش بخشی از نمره است.

        \item در سؤالاتی که طراحی یا مثال خواسته شده، ملموس‌ترین و واضح‌ترین
              شکل ممکن را بنویسید.

        \item نوشتن کد در این آزمون لازم نیست، اما اگر قطعه‌کدی در پاسخ آوردید
              باید از صحتش مطمئن باشید.
    \end{enumerate}

    \vspace{5mm}

    موفق باشید.
