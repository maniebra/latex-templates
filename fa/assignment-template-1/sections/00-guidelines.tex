\begin{neobox}[neotitlefont=\pixelfont]{قوانین، نکات و سیاست‌ها}

    \vspace{5mm}

    \begin{center}
        \Large{پیش از انجام تمرین حتماً نکات زیر را مطالعه فرمایید:}
    \end{center}

    \begin{enumerate}
        \item پیش از اقدام به انجام تمرین، حداقل یک بار اسلایدهای درس را مرور
              نمایید.

        \item استفاده از هوش مصنوعی برای تحقیق و جستجو مانعی ندارد، اما سپردن
              نوشتن پاسخ‌های نهایی به آن مجاز نیست و در صورت تشخیص، نمره‌ی
              سؤال مربوطه صفر در نظر گرفته می‌شود.

        \item هم‌فکری میان دانشجویان تشویق می‌شود، اما اشتراک‌گذاری پاسخ‌ها طبق
              آیین‌نامه‌های آموزشی ممنوع است و با هر دو طرف مطابق مقررات برخورد
              خواهد شد.

        \item تمرین سه بخش نظری (۳۵ درصد)، پژوهشی (۲۵ درصد) و عملی (۴۰ درصد)
              دارد و در مجموع \hwgrade{} نمره از نمره‌ی کل درس را تشکیل می‌دهد.
              برای دقت بیشتر، نمره‌ی این تمرین از ۱۰۰ لحاظ می‌شود.

        \item استفاده از کتاب‌ها، مقالات علمی و پروژه‌های
              متن‌باز\footnote{\lr{Open-source Projects}} برای پاسخ دقیق‌تر
              پیشنهاد می‌شود؛ ذکر منابع در انتهای پاسخ‌نامه انتظار می‌رود.

        \item سیاست جریمه‌ی تأخیر همان است که در ابتدای نیم‌سال تصویب شد.

        \item آنچه در سامانه بارگذاری می‌کنید باید یک فایل فشرده با قالب
              \lr{\texttt{.zip}} و ساختار زیر باشد:

              \begin{neocode}[neocodetitlefont=\pixelfont]{file\_format.txt}
@\coursecode@-HW@\hwnumber@-[STDID].zip
|
|--@\coursecode@-HW@\hwnumber@-[STDID].pdf
|--HW@\hwnumber@-Practical/
|       |
|       |-- *
              \end{neocode}

              که \lr{\texttt{STDID}} شماره‌ی دانشجویی شماست. فایل \lr{PDF}
              پاسخ بخش نظری به همراه گزارش بخش عملی را دارد و فایل‌های عملی در
              پوشه‌ی \lr{\texttt{HW\hwnumber\_Practical}} قرار می‌گیرند.

        \item جز در مواردی که صراحتاً قید شود، در انتخاب زبان و ابزار آزادید.
              با این حال در گزارش خود باید هر پکیج و کتابخانه‌ای را که استفاده
              کرده‌اید مختصراً توضیح داده و دلیل انتخابش را بیاورید.
    \end{enumerate}
