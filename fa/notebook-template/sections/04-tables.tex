\begin{note}[مشخصات ستون]
\lr{\texttt{X}} ستون کشسان است — متن را به آن بدهید و بگذارید بقیه به اندازه‌ی
محتوایشان جا بگیرند. \lr{\texttt{colspec = \{lXr\}}} پاسخگوی بیشتر جدول‌هایی
است که خواهید نوشت.
\end{note}

\section{فهرست‌ها}

نشانه‌ی گلوله در سطح اول رنگ تأکید می‌گیرد و در سطح‌های عمیق‌تر کم‌رنگ می‌شود:

\begin{itemize}
  \item آیتم سطح اول، با رنگ تأکید.
  \begin{itemize}
    \item سطح دوم، توخالی و کم‌رنگ.
    \begin{itemize}
      \item سطح سوم، یک خط تیره‌ی ساده.
    \end{itemize}
  \end{itemize}
  \item فاصله‌ها از پیش‌فرض‌های لاتک جمع‌تر شده‌اند.
\end{itemize}

فهرست‌های شماره‌دار عدد را با رنگ تأکید می‌گیرند:

\begin{enumerate}
  \item \lr{\texttt{vars/colors.tex}} و \lr{\texttt{vars/meta.tex}} را ویرایش کنید.
  \item در \lr{\texttt{sections/}} بنویسید.
  \item هر فایل را به \lr{\texttt{main.tex}} اضافه کنید.
  \item دستور \cmd{latexmk -xelatex main.tex} را اجرا کنید.
\end{enumerate}
