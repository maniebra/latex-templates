\chapter{کادرهای هشدار}

کادرها شکل اوبسیدین را دنبال می‌کنند: بدنه‌ی رنگ‌آمیزی‌شده، یک نوار پررنگ در
لبه‌ی سمت راست، و یک آیکون به همراه عنوان با رنگ خودِ کادر. این کادرها بین
صفحه‌ها می‌شکنند، پس یک کادر بلند صفحه‌آرایی را خراب نمی‌کند.

\section{مجموعه‌ی آماده}

\begin{note}
عنوان به‌طور پیش‌فرض همان نام کادر است. برای تغییرش آرگومان بدهید:
\lr{\texttt{\textbackslash begin\{note\}[چیز دیگری]}}.
\end{note}

\begin{tip}[درست انتخاب کنید]
چهارده نوع کادر آماده است. در یک نوشته بیش از سه یا چهار نوع به کار نبرید؛
تمام فایده‌ی کادر این است که به چشم بیاید، و وقتی هر بند یک کادر باشد این
فایده از بین می‌رود.
\end{tip}

\begin{warning}
کادرها داخل جدول و شناور بد رفتار می‌کنند. اگر لازمشان دارید، محتوا را داخل
کادر بگذارید و کادر را در جریان متن، نه برعکس.
\end{warning}

\begin{caution}[از دست رفتن داده]
دستور \cmd{git reset --hard} کارِ ثبت‌نشده را بدون بازگشت دور می‌ریزد. پیش از
اجرای آن \cmd{git stash list} را ببینید.
\end{caution}

\begin{example}[ساختن کادر تازه]
یک خط در \lr{\texttt{styles/callouts.sty}}:

\vspace{4pt}
\lr{\texttt{\textbackslash newcallout\{aside\}\{grape\}\{\textbackslash faIcon\{feather\}\}\{حاشیه\}}}
\end{example}

\begin{question}
این باید کادر باشد، پانوشت، یا یادداشت حاشیه؟ اگر بحث را قطع می‌کند، کادر.
اگر پشتیبانش است، پانوشت. اگر درباره‌اش نظر می‌دهد، حاشیه.
\end{question}

\begin{success}
هر چهارده کادر چیده می‌شوند، می‌شکنند، و یک سطح تودرتویی را تحمل می‌کنند.
\end{success}

\begin{quotebox}[برینگهرست]
صفحه‌ی حروف‌چینی‌شده نقشه‌ی ذهن است. نقشه‌ی جامعه‌ای هم هست که آن را ساخته.
\end{quotebox}

\section{حاشیه‌های بی‌عنوان}

وقتی رنگ و نوار را می‌خواهید اما سطر عنوان را نه، از \lr{\texttt{aside}}
استفاده کنید:

\begin{aside}[teal]
بی آیکون، بی عنوان — فقط یک بند نشانه‌گذاری‌شده. مناسب ترجمه، ارجاع به منبع،
یا تعریفی که می‌خواهید جدا بایستد بی‌آنکه خودش را جار بزند.
\end{aside}

بهتر خوانده می‌شود.
\end{idea}

\section{تمرین‌ها}

\lr{\texttt{exercise}} کادری دوتکه است: هرچه پیش از \lr{\texttt{\textbackslash tcblower}}
بیاید صورت پرسش است و هرچه پس از آن، پاسخ.

\begin{exercise}[مجموع تلسکوپی]
نشان دهید برای هر $n \geq 1$ داریم
$\sum_{k=1}^{n} \frac{1}{k(k+1)} = \frac{n}{n+1}$.
\tcblower
بنویسید $\frac{1}{k(k+1)} = \frac{1}{k} - \frac{1}{k+1}$. با جمع‌زدن، هر جمله
با همسایه‌اش حذف می‌شود جز اولی و آخری، و
$1 - \frac{1}{n+1} = \frac{n}{n+1}$ می‌ماند.
\end{exercise}

عنوان اختیاری است؛ بدون آن کادر فقط شماره می‌گیرد.

\begin{exercise}
ثابت کنید بزرگ‌ترین عدد اول وجود ندارد.
\tcblower
فرض کنید $p_1, \dots, p_n$ همه‌ی آن‌ها باشند. آنگاه $N = p_1 \cdots p_n + 1$
عاملی اول دارد که هیچ‌کدام از $p_i$ نیست، چون هر کدام باقیمانده‌ی ۱ می‌گذارند.
\end{exercise}

\subsection{جای خالی برای پاسخ}

وقتی پاسخ کارِ خواننده است نه کتاب، از \lr{\texttt{\textbackslash workspace}}
استفاده کنید — کادری خالی روی همان شبکه‌ی نقطه‌ای جلد.

\begin{exercise}[برای خواننده]
همه‌ی عددهای صحیح $n$ را بیابید که $n^2 + 1$ بر $n^3 + 1$ عاد کند.
\end{exercise}
