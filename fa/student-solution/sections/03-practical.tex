\section{سوال عملی \points{40}}

\begin{question}
    پیاده‌سازی کنید و گزارش کارتان را بنویسید.
\end{question}

\begin{answer}[پیاده‌سازی]
    توضیح کوتاهی درباره‌ی روش، بعد کد.

    \begin{code}{python}{solution.py}
    def solve(n):
        return n * (n + 1) // 2
    \end{code}

    با \cmd{python solution.py 10} اجرا می‌شود.
\end{answer}

\begin{answer}[نتیجه]
    خروجی چه بود و آیا با بخش نظری بالا می‌خواند.

    \begin{terminal}
$ python solution.py 10
55
    \end{terminal}
\end{answer}
