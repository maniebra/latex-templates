\begin{frame}{پنل‌ها}
    \begin{panel}{پنلی با عنوان}
        \begin{itemize}
            \item \lr{\texttt{panel}} عنوان را به عنوان آرگومان اجباری می‌گیرد.
            \item بین اسلایدها می‌شکند، پس فهرست بلند سرریز نمی‌کند.
            \item هر کلید \lr{\texttt{tcolorbox}} در آرگومان اختیاری می‌آید.
        \end{itemize}
    \end{panel}
\end{frame}

    \begin{panel}[height=2.6cm]{ارتفاع ثابت}
        با کلید \lr{\texttt{height}} پنل‌هایی را که کنار هم می‌نشینند هم‌اندازه کنید.
    \end{panel}

    \vspace{0.3cm}

    \begin{panel}[height=2.6cm]{پنل دوم}
        همه‌ی رنگ‌ها از \lr{\texttt{vars/colors.tex}} می‌آیند؛ یکی را آنجا عوض
        کنید و همه‌ی پنل‌ها تغییر می‌کنند.
    \end{panel}
\end{frame}
