\begin{frame}{Blocks and alerts}
    \begin{block}{Definition}
        Beamer's stock \texttt{block} is restyled as a window: tinted title
        bar, accent title, traffic lights.
    \end{block}

    \vspace{0.3cm}

    \begin{alertblock}{Careful}
        \texttt{alertblock} is the same box in the alert colour. Use it
        sparingly, or it stops meaning anything.
    \end{alertblock}
\end{frame}

\begin{frame}{Text and maths}
    \begin{panel}{Emphasis}
        Body text is Inter, headings are Lexend. \alert{Alerted text} picks up
        the red accent, \textbf{bold} and \emph{italic} work as usual.

        \vspace{0.2cm}
        Display maths sits on the panel background:
        \[
            v_{\text{avg}} = \frac{1}{n}\sum_{i=1}^{n} p_i
        \]
    \end{panel}
\end{frame}
